# ============================================================
#  PowerShell 配置文件 —— 让终端正确显示中文
#
#  为什么需要这个：
#    Windows 控制台默认代码页是 936（GBK）。
#    而 gcc / clang 编译出来的程序，字符串按 UTF-8 存。
#    GBK 终端去解码 UTF-8 字节 -> 中文乱码。
#
#    症状举例：
#        学校 C 语言课 · 作业环境正常
#      变成
#        瀛︽牎 C 璇█璇? 浣滀笟鐜璇娉ｅ父
#
#  已实测：
#    代码页 936 时运行 -> 乱码
#    先 chcp 65001 再运行 -> 完全正常
#
#  所以这里在每次打开终端时自动切到 UTF-8。
#
#  影响范围：
#    仅影响【新的终端窗口】，且只影响编码，不影响其他功能。
#    如果哪天想临时切回 GBK，执行： chcp 936
#
#  文件必须存为 UTF-8 with BOM，否则 PowerShell 5.1 会把中文注释读成乱码。
# ============================================================

try {
    # 控制台代码页切到 UTF-8（这是修复乱码的关键一步）
    $null = & chcp.com 65001 2>&1

    # 让 PowerShell 自己读写也用 UTF-8
    [Console]::OutputEncoding = [System.Text.Encoding]::UTF8
    [Console]::InputEncoding  = [System.Text.Encoding]::UTF8

    # 管道传输也用 UTF-8（防止 | Out-File 之类写出乱码）
    $OutputEncoding = [System.Text.Encoding]::UTF8
}
catch {
    # 任何一条失败都不该阻止终端启动，静默忽略
}
